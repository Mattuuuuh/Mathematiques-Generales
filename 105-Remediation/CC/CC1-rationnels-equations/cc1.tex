\exe{4}{
	Exprimer le nombre $q$ suivant sous forme de fraction d'entiers.
		\[ q = \frac23 \times \frac{~~5~~}{\frac12} + \frac56 + 1. \]

	\fillwithlines{4cm}
}{exe:Q}{
	\begin{align*}
		q &= \frac23 \times 5 \times \frac21 + \frac56 + 1 \\
		&= \frac{20}3 + \frac56 + 1 \\
		&= \frac{40}6 + \frac56 + \frac66 \\
		&= \frac{51}6
	\end{align*}
}

\exe{4}{
	Matthieu passe un tiers de sa vie à dormir.
	Lorsqu'il est éveillé, il passe $\frac25$ de son temps à travailler. Le reste du temps, il lit des livres.
	
	\begin{enumerate}
		\item
		Quelle proportion de son temps Matthieu passe-t-il à lire ?
		\item
		En une semaine (7 jours de 24 heures), combien d'heures Matthieu passe-t-il à lire ?
		Une valeur numérique (avec virgule) est attendue.
	\end{enumerate}

	\fillwithlines{7cm}
}{exe:contexte}{
	\begin{enumerate}
		\item
		Le temps passé éveillé correspond à $1-\frac13 = \frac23$ de la vie de Matthieu.
		Comme $\frac25$ du temps éveillé est passé à travailler, $1-\frac25 = \frac35$ du temps éveillé est passé à lire.
		Ceci correspond à $\frac35\times\frac23 = \frac25$.
		\item
		Il s'agit de calculer $\frac25$ de $7\times24$.
		Pour diviser par 5, il suffit de multiplier par 2 et de diviser par 10.
			\[ \frac25 \times 7\times 24 = \frac{336}5 = \frac{672}{10} = 67,2. \]
		La réponse attendue est donc $67,2$ heures.
	\end{enumerate}
}

\exe{2}{
	Donner le nombre rationnel $x\in\Q$ vérifiant
		\[ 3-4x = 7x + 1. \]

	\fillwithlines{4cm}
}{exe:eq-lin}{
	\[ 3-4x = 7x + 1 \iff 3-1 = 7x+4x \iff 2 = 11x \iff x=\frac2{11} \]
}

\exe{3}{
	Donner le nombre rationnel $x\in\Q$ différent de 2 vérifiant
		\[ \frac3{x-2} + 1 = 2-\frac7{x-2}. \]

	\fillwithlines{4cm}
}{exe:eq-inv}{

		\[ \frac3{x-2} + 1 = 2-\frac7{x-2} \iff \frac{10}{x-2} = 1 \iff 10 = x-2 \iff x = 12 \]
}

\exe{2}{
	Développer et réduire l'expression factorisée suivante.
		\[ \left(3x^2 - 5\right)^2 \]

	\fillwithlines{4cm}
}{exe:calcul-litteral}{
	En utilisant $(a-b)^2 = (a)^2 + (b)^2 - 2(a)(b)$, par exemple, nous obtenons
		\[ \left(3x^2 - 5\right)^2  = 9x^4 + 25 - 30x^2. \]
}

\exe{2}{
	Donner toutes les solutions $x\in\R$ de l'équation suivante.
		\[ (7x + 1)(2-x) = 0. \]

	\fillwithlines{4cm}
}{exe:prod-zero}{
	Le produit nul implique $7x+1 = 0 \iff x = -\frac17$ ou $2-x = 0 \iff x = 2$.
}

\exe{3}{
	Donner toutes les solutions $x\in\R$ de l'équation suivante.
		\[ 7x^2 = 3. \]
	Une réponse de la forme $q \sqrt{r}$ est attendue, où $q\in\Q$ est un rationnel et $r\in\N$ un entier naturel.

	\fillwithlines{5cm}
}{exe:sqrt}{
		\[ 7x^2 = 3 \iff x^2 = \frac37 \iff |x| = \sqrt{\frac37} \iff x = \pm \sqrt{\frac37}. \]
	Comme $\sqrt{\frac37} = \frac{\sqrt3}{\sqrt7} = \frac17 \sqrt{21}$, nous obtenons les deux solutions
		\[ \left\{ \frac17\sqrt{21} ; - \frac17\sqrt{21} \right\} \] 
}
