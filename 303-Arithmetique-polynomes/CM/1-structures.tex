\section{Groupe}

\dfn{groupe}{
	Un \emphindex{groupe} $\bigpar{G, \cdot}$ est un ensemble muni d'une loi satisfaisant les axiomes suivants.
	\begin{enumerate}%[label={\bf A\arabic*.}]
		\item
		\textbf{Loi interne.} Pour tout $g, h\in G$, $g\cdot h \in G$.
		\item
		\textbf{Loi associative.} Pour tout $g, h, k \in G$, $(g\cdot h)\cdot k = g \cdot (h \cdot k)$.
		\item
		\textbf{Élément neutre.} Il existe un élément $e\in G$ neutre pour la loi : pour tout $g\in G$, $g \cdot e = e \cdot g = g$.
		\item
		\textbf{Inverse.} Tout élément est inversible : pour tout $g\in G$, il existe $g^{-1} \in G$ tel que $g \cdot g^{-1} = e = g^{-1} \cdot g$.
	\end{enumerate}
}{dfn:groupe}

\nt{
	Dans un groupe, l'inverse à gauche est égal à l'inverse à droite.
}

\notation{
	Il est fréquent d'omettre la loi dans un groupe, celle-ci étant unique. 
	Le produit $g \cdot h$ sera alors noté $gh$.
}

\dfn{commutativité}{
	Une loi $\cdot$ sur un ensemble $E$ est \emphindex{commutative} dès que
		\[ a \cdot b = b \cdot a \]
	pour tous les éléments $a, b \in E$.
}{dfn:commutativite}

\nomen{
	Une loi $\cdot$ commutative est généralement notée $+$, par analogie à l'addition qui est toujours commutative, alors que la multiplication ne l'est parfois pas (dans le cas des matrices, par exemple).
}

\exe{}{
	Montrer que $\bigpar{ \Z ,+}$ est un groupe mais que $\bigpar{\Z, \cdot}$ n'en est pas un.
}{exe:Z-group}{
	Les axiomes sont tous vérifés pour la loi $+$.
	L'axiome de l'inverse n'est pas vérifé pour la loi $\cdot$.
}

\exe{}{
	Soit $f$ une fonction réelle sur $\R$.
	Montrer que l'ensemble des périodes de $f$ suivant est un sous-groupe de $\bigpar{\R, +}$.
		\[ P_f = \bigset{ p \in \R \tq f(x+p) = f(x) \forall x\in\R } \]
}{exe:groupe-periodes}{

}

\exe{}{
	Est-ce que l'ensemble $\bigset{ -10 ; -9 ; \dots ; 9 ; 10}$ est un groupe lorsque muni de $+$ ?
}{exe:loi-non-interne}{
	Non car la loi n'est pas interne.
}

\section{Anneau et corps}

\subsection{Anneau}

\dfn{anneau}{
	Un \emphindex{anneau} $\bigpar{ R, +, \cdot} $ est un ensemble ayant deux opérations satisfaisant les axiomes suivants.
	\begin{enumerate}
		\item
		$\bigpar{R, +}$ est un groupe abélien.
		\item
		La loi $\cdot$ est associative : pour tout $a, b, c\in R$, $a \cdot (b \cdot c) = (a \cdot b) \cdot c$.
		\item
		La loi $\cdot$ se distribue sur la loi $+$ : pour tout $a, b, c\in R$, 
		\begin{itemize}[label=$\bullet$]
			\item $(a+b) \cdot c = a \cdot c + b \cdot c$ ; et
			\item $a \cdot (b+c) = a \cdot b + a \cdot c$.
		\end{itemize}
	\end{enumerate}
}{dfn:anneau}

\nt{
	Attention : la structure $\bigpar{ R, \cdot}$ n'admet pas forcément d'élément neutre. Une telle structure avec élément neutre s'appelle un \emphindex{monoïde}.
}

\nomen{
	Un anneau $R$ est dit
	\begin{enumerate}%[leftmargin=80pt]
		\item[\emphindex{unitaire}] s'il existe $1\in R$ élément neutre de la loi  $\cdot$.
		$1 \cdot u = u \cdot 1 = u$ pour tout $u\in R$.
		\item[\emphindex{commutatif}] si la loi $\cdot$ est commutative.
		$a \cdot b = b \cdot a$ pour tout $a, b\in R$.
	\end{enumerate}
}

\ex{}{\leavevmode
	\begin{enumerate}
		\item
		$R = \Z$ est un anneau commutatif unitaire.
		\item
		$R = 2\Z$ est un anneau commutatif non unitaire.
		\item
		$R = \{ 0 \}$ est un anneau commutatif unitaire en identifiant $1=0$.
		\item
		$R = \Z[i] = \bigset{ a + bi \tq a, b\in\Z}$, les \emphindex{entiers de Gauss}, est un anneau commutatif unitaire.
		\item
		$R = \Z^{n \times n}$ et $\R^{n\times n}$ sont des anneaux unitaires non commutatifs.
		\item
		$R = C^{0}(\R) = \bigset{ f : \R \rightarrow \R \tq \text{$f$ continue} }$ est un anneau commutatif unitaire.
	\end{enumerate}
}{ex:anneaux}

\nt{
	L'élément 0 étant défini pour l'addition, il n'est pas clair \apriori~ que multiplier par 0 donne 0.
	L'élément 1 étant défini pour la multiplication, il n'est pas clair que multiplier par son inverse additif aboutit à l'inverse additif du facteur.
	
	La proposition suivante permet de déduire des axiomes de l'anneau unitaire les propriétés intuitives de 0 et de 1.
}

\prop{propriétés de $0$ et $1$}{
	Soit $R$ un anneau unitaire. Alors, pour tout $u\in R$,
	\begin{enumerate}
		\item $u \cdot 0 = 0 \cdot u = 0$ ; et
		\item $u \cdot (-1) = (-1) \cdot u = -u$.
	\end{enumerate}
}{prop:prop-0-1}

\exe{}{
	Montrer que $u \cdot 0 = 0$ pour tout $u$ élément d'un anneau unitaire $R$.
}{exe:utimes0}{
	Nous avons la suite d'égalités suivantes.
		\begin{align*}
			u \cdot 0 &= u \cdot (0+0)  && \text{car $\bigpar{R, +}$ est un groupe d'élément neutre $0$} \\
				&= u \cdot 0 + u \cdot 0 && \text{par associativité de $ \cdot $ sur $+$} \\
			0 &= u \cdot 0 && \text{en soustrayant $u \cdot 0$}
		\end{align*}
}

\exe{}{
	Montrer que $u \cdot (-1) = -u$ pour tout $u$ élément d'un anneau unitaire $R$.
}{exe:utimes-1}{
	Nous avons la suite d'égalités suivantes.
		\begin{align*}
			u \cdot (-1) + u &= u \cdot (-1+1) && \text{par associativité de $ \cdot $ sur $+$} \\
				&= u \cdot 0 && \\
				&= 0 &&
		\end{align*}
}

\dfn{unité, ensemble des unités}{
	Un élément $u\in R$ est dit \emphindex{unité} de l'anneau $R$ unitaire s'il existe $v\in R$ tel que
		\[ u \cdot v = 1 = v \cdot u. \]
	L'\emphindex{ensemble des unités} est donné par
		\[ R^\times = \bigset{ u \in R \tq \text{$u$ est unité} }. \]
}{dfn:unite}

\nomen{
	Une unité est aussi appelé un élément \emphindex{inversible}.
}

\exe{}{
	Montrer que $\bigpar{ R^\times,  \cdot  }$ est un groupe.
}{exe:units-group}{
	todo
}

\exe{}{
	Soit $R$ un anneau unitaire.
	Montrer que si $u\in R$ admet un inverse à gauche et un inverse à droite, alors ces inverses sont égaux.
		\[ u^G \cdot u = 1 = u \cdot u^D \implies u^G = u^D. \]
}{exe:inv-gauche-droite}{
	Par associativité, $u^D = (u^G u) u^D = u^G (u u^D) = u^G$.
}

\exe{}{
	Considérons $\R[x]$, l'ensemble des polynômes à coefficients réels, comme $\R$-espace vectoriel.
	Soit $R = \text{End}(\R[x])$ le groupe des endomorphismes de $\R[x]$ muni de l'opérateur de composition.
	\begin{enumerate}
		\item
		Montrer que l'élément $m_x : p(x) \mapsto x\cdot p(x)$ admet un inverse à gauche mais pas à droite.
		\item
		Montrer que l'élément $\partial : p(x) \mapsto p'(x)$ admet un inverse à droite mais pas à gauche.
		\item
		Donner un élément inversible de $R$ qui n'est pas l'élément neutre.
	\end{enumerate}
}{exe:inv-gauche-not-droite}{
	Il y a en fait une infinité d'inverses à gauche et à droite !
}

\subsection{Corps}

\dfn{corps}{
	Soit $R$ un anneau commutatif unitaire dans lequel on suppose $1\neq0$.
	$R$ est un \emphindex{corps} si $R^\times = R-\{0\}$.
	Chaque élément du corps, hormis 0, est inversible.
}{dfn:corps}

\notation{
	Nous noterons $\K$ les corps dans la suite.
}

\exe{}{
	Montrer que si $1=0$ dans un anneau $R$ commutatif unitaire, alors $R = \bigset{ 0 }$.
	
	Montrer que si $0$ est inversible, alors $1=0$.
}{exe:1-is-0}{
	Tout élément $u$ de $R$ vérifie $u = u \cdot 1 = u \cdot 0 = 0$.
	
	Si $0$ est inversible, alors $0 = 0 \cdot 0^{-1} = 1$.
}

		Montrer que $\circ$ est associative d'élément neutre 0.
		\item

	\end{enumerate}
}{exe:circle-op}{

\dfn{corps gauche}{
	Soit $R$ un anneau unitaire non commutatif dans lequel on suppose $1\neq0$.
	$R$ est un \emphindex{corps gauche} si $R^\times = R-\{0\}$.
	Chaque élément du corps, hormis 0, est inversible.
}{dfn:corps-gauche}

\ex{}{
	L'ensemble $\H$ des \emphindex{quaternions} défini par
		\[ \H = \bigset{ a+bi+cj+dk \tq a, b, c, d \in \R, i^2 = j^2 = k^2 = -1, \et ij = k, jk = i, ki = j} \]
	est un corps gauche.
	
	En effet, $(a+bi+cj+dk)^{-1} = \frac{a-bi-cj-dk}{a^2 + b^2 + c^2 + d^2}$ si $(a,b,c,d)\neq(0,0,0,0)$
	et $ij = (jk)(ki) = jk^2 i = -ji$.
	
	Il s'avère que les quaternions traduisent des rotations dans $\R^4$ et s'identifient à un ensemble de matrices $4\times4$.
	Leur non commutativité n'est donc pas surprenante.
}{ex:quaternions}

\subsection{Anneau intègre}

\nomen{
	Un élément non nul $u\in R-\{0\}$ d'un anneau $R$ est un \emphindex{diviseur de zéro} s'il existe $v\in R-\{0\}$ non nul également tel que $u \cdot v = 0$.
}

\exe{}{
	Soit $n\in\N^*$ et $R = \Z/n\Z$ l'anneau des entiers modulo $n$.
	Donner l'ensemble des diviseurs de zéro et l'ensemble des éléments inversibles de $R$.
}{exe:ZnZ-div-zero}{
	Nous allons montrer que $R$ se partionne comme
		\[ R = \{ 0 \} \sqcup R^\times \sqcup \bigset{ \text{ diviseurs de zéro } }. \]
	D'abord, un élément $x\in R-\{0\}$ est diviseur de zéro si et seulement si $d = \pgcd(x,n) \neq 1$ dans $\Z$.
	En effet, si $x$ divise zéro, alors il existe un $y \in R-\{0\}$ tel que $xy = 0$.
	Ainsi, $n$ divise le produit $xy$ mais $n$ ne divise pas $y$.
	Ainsi $d\neq1$, sinon le lemme d'Euclide impliqerait $n|y$.
	
	Réciproquement, si $d \neq 1$, alors $x \times \frac{n}d = \frac{x}d n = 0$ dans $R$, et $\frac{n}d \neq 0$.
	
	D'abord, un élément $x\in R-\{0\}$ est inversible si et seulement si $d = \pgcd(x,n) = 1$ dans $\Z$.
	Si $x$ est inversible, il existe $y\in R$ tel que $xy = 1$.
	Nous obtenons une identité de Bézout : $xy + un = 1$ pour un certain $u\in\Z$.
	Comme le PGCD $d$ divise $x$ et $n$ dans $\Z$, il divise donc aussi 1, et $d=1$.
	
	Réciproquement, si $d=1$. alors l'identité de Bézout $xu + nv = 1$ avec $u, v \in \Z$ implique que $xu = 1$ dans $R$ et que $x$ est unité.
}

\exe{}{
	Considérons $R = \bigset{ f : \R \rightarrow \R }$ l'ensemble des applications de  $\R$ dans $\R$.
	\begin{enumerate}
		\item Donner les éléments neutres pour l'addition et la multiplication dans l'anneau $\bigpar{R, +, \cdot}$.
		\item Décrire les diviseurs de zéro de $R$.
		\item Montrer que chaque élément non nul est soit un diviseur de zéro, soit inversible.
	\end{enumerate}
}{exe:ring-of-maps}{
	todo
}

\prop{}{
	Soit $R$ un anneau unitaire et $u\in R$ une unité.
	Alors $u$ ne divise pas zéro.
}{prop:unit-dont-divide-zero}

\pf{}{
	Si $u$ divise zéro, alors il existe $v\neq0$ tel que $u\cdot v = 0$.
	Comme $u$ est unité, nous avons $v = (u^{-1} u) v = u^{-1} (uv) = u^{-1} 0 = 0$.
	Ceci est contradictoire.
}

\nomen{
	Un anneau $R$ commutatif unitaire est $\emphindex{intègre}$ dès qu'il n'admet aucun diviseur de zéro.
}

\exe{}{
	Montrer que la réciproque de la proposition \ref{prop:unit-dont-divide-zero} est fausse.
	%Au regard de l'exercice \ref{exe:ZnZ-div-zero}, considérer un anneau de caractéristique nulle.
}{exe:notunit-nordivzero}{
	L'élément 2 de l'anneau $\Z$ n'est pas une unité ni un diviseur de zéro.
}

\cor{}{
	Tout corps $\K$ est intègre : si $u \cdot v = 0$ dans $\K$, alors $u = 0$ ou $v = 0$.
}{cor:corps-integre}

\pf{}{
	Si $u=0$, il n'y a rien à démontrer.
	Sinon, comme $\K$ est un corps, $u$ est inversible.
	Donc $v =  (u^{-1} u ) v = u^{-1} (uv) = u^{-1} 0 = 0$.
}

\subsection{Idéal}

\dfn{idéal}{
	Soit $R$ un anneau et $I\subset R$ un ensemble.
	\begin{enumerate}
		\item
		$I$ est un \emphindex{idéal à gauche} si
		\begin{enumerate}
			\item
			$\bigpar{I, +}$ est un groupe abélien ; et
			\item
			$r \cdot  I \subset I$ pour tout $r\in R$.
		\end{enumerate}
		\item
		$I$ est un \emphindex{idéal à droite} si
		\begin{enumerate}
			\item
			$\bigpar{I, +}$ est un groupe abélien ; et
			\item
			$I \cdot r \subset I$ pour tout $r\in R$.
		\end{enumerate}
		\item
		$I$ est un \emphindex{idéal} ou \emphindex{idéal bilatère} s'il est un idéal à gauche et à droite.
	\end{enumerate}
}{dfn:ideal}

\nt{
	Attention : lorsque non précisé « à gauche » ou « à droite », un idéal est donc un idéal bilatère.
}

\nt{
	Un idéal est un sous-anneau non unitaire qui est stable par multiplication extérieure.
}

\notation{
	L'expression « $I$ est un idéal de $R$ » est notée $I \lhd R$.
}

\exe{}{
	Soit $I, J \subseteq R$ deux idéaux d'un anneau $R$.
	Montrer que les ensembles suivants sont également des idéaux de $R$ et que chaque ensemble est inclus dans le suivant.
	\begin{enumerate}
		\item l'idéal produit $I\cdot J = \bigset{ a_1 b_1 + \dots + a_n b_n \tq a_i \in I, b_i \in J, n\in\N^* }$ ;
		\item l'idéal intersection $I\cap J$ ;
		\item l'idéal somme $I+J = \bigset{ a + b \tq a\in I, b\in J}$.
	\end{enumerate}
}{exe:somme-produit-inter-ideaux}{
	todo
}

\prop{}{
	Soit $I\lhd R$ un idéal d'un anneau. Alors
	\begin{enumerate}[label=(\roman*)]
		\item
		$I=R$ si et seulement si $I$ contient une unité.
		\item
		si $R$ est commutatif, $R$ est un corps si et seulement si ses seuls idéaux sont $\{ 0 \}$ et $R$.
	\end{enumerate}
	L'étude des idéaux n'est donc pas très intéressante dans un corps !
}{prop:ideaux-corps}


\dfn{idéal généré par un ensemble}{
	Soit $R$ un anneau et $A \subseteq R$ un sous-ensemble.
	L'\emphindex{idéal engendré par $A$}, noté $\japb{A}$, est le plus petit idéal de $R$ contenant $A$.
}{dfn:ideal-gen}

\prop{}{
	Soit $R$ un anneau et $A \subseteq R$ un sous-ensemble.
	Alors
		\[ \japb{A} = \Bigset{ \sum_{i=1}^n r_i a_i r_i' \tq r_i, r_i' \in R, a_i \in A, n\in\N^*}. \]
	Si $R$ est commutatif, alors
		\[ \japb{A} = \Bigset{ \sum_{i=1}^n r_i a_i \tq r_i \in R, a_i \in A, n\in\N^*}. \]
}{prop:ideal-gen}

\nomen{
	Un idéal est de \emphindex{type fini} s'il est généré par un ensemble fini.
	Un idéal généré par un unique élément est \emphindex{principal} et noté $I = \japb{a} = (a)$.
}

\subsection{Divisibilité et irréductibilité}

\dfn{divisiblité}{
	Soient deux éléments $a, b \in R$ d'un anneau intègre (donc commutatif, unitaire, et sans diviseur de zéro).
	L'élément $a$ \emphindex{divise} $b$ dans $R$ dès que
		\[ b = a\cdot q, \]
	où $q\in R$.
}{dfn:divisibilite}

\notation{
	La relation « $a$ divise $b$ » est notée $a | b$.
}

\prop{}{
	Soient $a, b\in R$. Alors
		\begin{align*}
			a | b && \iff && \japb{b} \subseteq \japb{a}.
		\end{align*}
}{prop:incl-is-div}

\nomen{
	Un diviseur non unité est appelé un \emphindex{diviseur propre}.
	
	Deux éléments qui sont multiples l'un de l'autre par un unité sont \emph{associés}.
}

\exe{}{
	Montrer que $u$ est un diviseur propre de $v$ si et seulement si $\japb{v} \subset \japb{u}$, l'inclusion étant stricte.
}{exe:div-propre-ideal}{
	todo
}

\exe{}{
	Montrer que $u, v\in R$ sont associés si et seulement si $\japb{a} = \japb{b}$.
}{exe:associe-ideal}{
	todo
}

\dfn{irréductible}{
	Un élément $u\in R$ non nul est \emphindex{irréductible} dès que
		\begin{enumerate}
			\item $u$ n'est pas unité ; et
			\item $u$ n'admet aucun diviseur propre.
		\end{enumerate}
}{dfn:irreductible}

\prop{}{
	Soit $R$ un anneau intègre.
	\begin{enumerate}
		\item
		Soient $r\in R$ irréductible et $u\in R$ unité. Alors $u\cdot r$ est irréductible.
		\item
		Soient $r, s\in R$ deux irréductibles. Alors
		\begin{align*}
			r | s && \iff && r | s && \iff && r = u\cdot s, \text{ avec $u$ une unité de $R$.}
		\end{align*}
	\end{enumerate}
	Une décomposition en facteurs irréductibles ne peut donc être unique qu'à multiplication de chaque facteur par une unité près.
}{prop:up-to-unit}

\pf{}{
	\begin{enumerate}
		\item
		Soient $r\in R$ irréductible et $u\in R$ unité. Montrons que $u\cdot r$ est irréductible.
		D'une part, $u \cdot r$ n'est pas une unité, car $R^\times$ est un groupe : si $u \cdot r \in R^\times$, alors $r \in u^{-1} R^\times = R^\times$.
		
		D'autre part, si $u\cdot r = a \cdot b$ dans $R$, alors $r = (u^{-1}a) \cdot b$.
		Donc soit $b$ est unité, soit $u^{-1}a$ est unité, et donc $a$ est unité car $R^\times$ est un groupe à nouveau.
		\item
		Soient $r, s\in R$ deux irréductibles. Alors $s|r$ est équivalent à l'existence d'un $u\in R$ tel que $r = u \cdot s$.
		Comme $r$ n'est pas unité, $u$ l'est.
		Nous avons aussi $s = u^{-1}r$ et $r|s$.
	\end{enumerate}
}

\subsection{Anneaux factoriels à travers l'exemple de $\Z$}

\dfn{anneau factoriel}{
	Un anneau intègre (donc commutatif, unitaire, et sans diviseur de zéro) $R$ est un \emphindex{anneau factoriel} si tout élément $r$ non nul et non inversible se décompose \underline{de façon unique} comme produit d'irréductibles
		\[ r = p_1 \cdot p_2 \cdot \cdots \cdot p_k, \]
	à multiplication par une unité et à permutation des facteurs près.
}{dfn:anneau-factoriel}

\thm{décomposition en facteurs premiers}{
	L'anneau $\Z$ est factoriel.
	
	Soit $n\in\Z$ un entier relatif. Supposons $n\neq0$ (non nul) et $n\neq \pm1$ (non inversible).
	Alors $n$ se décompose de façon unique comme
		\[ n = p_1 \cdot p_2 \cdot \cdots \cdot p_k, \]
	où les $p_i$ prennent la forme $\pm p$ avec $p\in\N$ un nombre premier, et à permutation près.
}{thm:decomp-premiers}

\pf{}{
	Le plan classique de la démonstration est décrit ici.
	Chaque propriété est essentielle et découle de la propriété précédente.
	Ce plan fera l'objet de la même considération dans l'anneau $\K[x]$ section \ref{sec:Kx} du chapitre \ref{chap:polynomes}.
	
	Dans l'anneau $\Z[x]$ étudié section \ref{sec:Zx} du chapitre \ref{chap:polynomes}, seules les deux dernières propriétés sont accessibles et nécessitent donc une stratégie particulière.
	
	\begin{enumerate}[label=\bf\underline{Propriété \arabic*.}]
		\item
		L'anneau $\Z$ est équipé d'une division euclidienne.
			\[ \forall a, b\in\Z, \exists q, r\in\Z \tq b = aq + r, \et |r| < |a|. \]
		\item
		Le théorème de Bézout énonce que 
			\[ \pgcd(a, b)=1 \iff \exists u, v\in\Z \tq au + bv = 1. \]
		\item
		%Classification des éléments irréductibles.
		Dans $\Z$, les éléments irréductibles sont exactement les $\pm p$ avec $p\in\N$ premier.
		\item
		Le lemme d'Euclide dit que si $p\in\N$ est irréductible, alors
			\[ p|(ab) \implies p|a \ou p|b. \]
	\end{enumerate}
	
	Considérons $n\in\Z$ et démontrons le théorème \ref{thm:decomp-premiers} en admettant la véracité de chaque propriété. 
	
	\underline{Existence.}
	
	Si $n$ n'est pas déjà irréductible, alors réduire $n$, et répéter la réduction de chaque facteur.
	Ce processus se termine nécessairement en au plus $n$ étapes car la valeur absolue d'un diviseur prope de $n$ est positive et strictement inférieure à celle de $n$.
	
	\underline{Unicité.}
	
	Considérons deux décompositions
		\[ n = p_1 \cdot p_2 \cdot \cdots \cdot p_k = q_1 \cdot q_2 \cdot \cdots \cdot q_l. \]
	Alors $p_1$ divise l'un des facteurs $q_i$ d'après le lemme d'Euclide et, par irréducibilité, $p_1 = \pm q_i$.
	Par le même raisonnement, chaque irréductible $p_i$ est égal à un $q_j$ à multiplication par unité près.
	Par symétrie en $p$ et $q$, les deux décompositions sont égales à permutation et à multiplication par une unité près.
}

\section{Algèbre sur un corps}

\motiv{
	Considérons le polynôme $p(x) = x^3 + x^2 - 3x + 1 \in \R[x]$ et demandons-nous : pour quel objet $a$ est-ce que l'évaluation $p(a)$ a un sens ?
	Notons $A$ l'ensemble des objets et étudions ce que $A$ doit admettre comme structure.
	Il faut pouvoir
	\begin{enumerate}
		\item
		multiplier des objets de $A$ ensemble, car $a^2 =a \times a$ doit avoir un sens ;
		\item
		additionner des objets de $A$ ensemble, car $a^3 + a^2$ doit avoir un sens ;
		\item
		multiplier chaque objet de $A$ par un nombre réel du corps $\R$, car $-3x = (-3)\cdot a$ doit avoir un sens.
	\end{enumerate}
	Remarquons que nous devons avoir trois loi.
	Deux lois sont internes à $A$, $+$ et $\times$, et une loi est externe de $\R$ sur $A$.
	Les deux premières appellent une structure d'anneau, et la deuxième celle d'espace vectoriel.
	
	Pour bien évaluer le polynôme $x^3 + x^2 - 3x + 1\in\R[x]$ en un objet $a$ de $A$, il conviendra alors de voir $1$ comme $1\cdot1$, où le premier $1$ est dans $\R$, et le deuxième $1$ dans $A$.
	
	Il est en outre souhaitable que $p(x) + q(x) = (p+q)(x)$ et que $p(x)\times q(x) = (p\cdot q)(x)$.
	Pour ce faire, les lois de $A$ doivent être assiociatives, les multiplications ($\cdot$ et $\times$) doivent se distribuer sur l'addition $+$, et les loi $\cdot$ et $\times$ doivent être compatibles : $(k\cdot x)\times(k' \cdot x') = (kk')(x\cdot x')$.
	
	Notons qu'il n'est pas nécessaire que l'algèbre soit commutative pour avoir $p(a)\times q(a) = (p\cdot q)(a)$ car les puissances de $a$ commutent toujours entre elles.
	Un exemple est fourni à l'exercice \ref{exe:verif-R2-algebra}.
	
	Pour alléger les notations, nous avons caché une quatrième et une cinquième loi, celles de l'addition et de la multiplication internes au corps $\K$.
}

\ex{}{
	Considérons $p(x) = x^3 + x^2 - 3x + 1\in\R[x]$.
	Pour chaque ensemble $A$ et élément $a\in A$ suivants, nous décrivons trois lois $+, \cdot, \times$ sur $\R$ et calculons $p(a)$.
	\begin{enumerate}
		\item 
		$A = \R, a = 3$. Munissons $A$ des loi d'addition et de multiplication usuelles.
		Alors 
			\[ p(a) = 3^3 + 3^2 - 3\cdot 3 + 1\cdot1 = 28. \]
		\item
		$A = \C, a = 1+i$. Munissons $\C$ des lois $+$ et $\cdot$ usuelles comme $\R$-espace vectoriel :
			\begin{align*}
				(a+bi) + (c+di) = (a+b) + (c+d)i && \et &&  k\cdot (a+bi) = (ka) + (kb) i.
			\end{align*}
		Ajoutons la multiplication interne usuelle des nombres complexes
			\[ (a+bi) \times (c+di) = (ac - bd) + (ad + bc)i. \]
		Alors 
			\[ p(a) = (1+i)^3 + (1+i)^2 - 3\cdot(1+i) + 1\cdot1 = -4 + i. \]
		\item
		$A = \R^{2\times2}, a = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$.
		Munissons $\R^{2\times2}$ des lois $+$ et $\cdot$ usuelles comme $\R$-espace vectoriel :
			\begin{align*}
				\begin{pmatrix} a & b \\ c & d\end{pmatrix} + \begin{pmatrix} a' & b' \\ c' & d'\end{pmatrix}  = \begin{pmatrix} a+a' & b+b' \\ c+c' & d+d' \end{pmatrix}  && \et &&  k\cdot\begin{pmatrix} a & b \\ c & d\end{pmatrix}  = \begin{pmatrix} ka & kb \\ kc & kd \end{pmatrix}.
			\end{align*}
		Ajoutons la multiplication interne usuelle des matrices
			\[ \begin{pmatrix} a & b \\ c & d\end{pmatrix} \times \begin{pmatrix} a' & b' \\ c' & d'\end{pmatrix}  = \begin{pmatrix} aa' + bc' & ab'+bd' \\ ca'+dc' & cb'+dd' \end{pmatrix}.\]
		Alors
			\[ p(a) = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}^3 + \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}^2 - 3\cdot\begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix} + 1\cdot\begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix} = \begin{pmatrix} 42 & 58 \\ 87 & 129 \end{pmatrix} \]
	\end{enumerate}
}{ex:algebres}

\exe{}{
	En reprenant l'exemple \ref{ex:algebres}, vérifier que $q(x) = (x-1)(x^2 + 2x - 1)$ est égal à $p(x)$ dans $\R[x]$ puis qu'on a bien $q(a) = p(a)$ pour $a = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$.
	
}{exe:verif-R2-algebra}{
	En remplaçant le $1$ par l'unité de l'algèbre $I$, nous devons calculer $q(a) = (a-I)(a^2 + 2a - I)$.
	
	D'abord, 
		\[a - I = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix} - \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix} = \begin{pmatrix} 0 & 2 \\ 3 & 3 \end{pmatrix}.\]
	Puis 
		\[ a^2 + 2a - I = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix} + 2\begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix} - \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix} = \begin{pmatrix} 8 & 21 \\ 14 & 29 \end{pmatrix}.\]
	Finalement, 
		\[ q(a) = (a-I)(a^2 + 2a - I) = \begin{pmatrix} 0 & 2 \\ 3 & 3 \end{pmatrix} \begin{pmatrix} 9 & 21 \\ 14 & 30 \end{pmatrix} = \begin{pmatrix} 42 & 58 \\ 87 & 129 \end{pmatrix}, \]
	ce qui est égal au $p(a)$ calculé dans l'exemple \ref{ex:algebres}.
	
	Remarquons d'ailleurs que la décomposition $p(x) =(x^2 + 2x - 1)(x-1)$ donne \apriori~un résultat différent lorsque évalué en $a$.
	Cependant, ce n'est pas le cas car les matrices $a-I$ et $a^2 + 2a -I$ commutent en tant que des puissances de $a$.
}

\nt{
	La factorisation dans $\R[x]$ garde son utilité lorsque $x$ est remplacé par un élément $a$ d'un anneau, même non commutatif. 
	Les puissances de $a$ commutent toujours entre elles.
}

\dfn{algèbre sur un corps}{
	Une \emphindex{algèbre sur un corps} $\bigpar{A, +, \cdot, \times}$ est un ensemble muni de deux lois satisfaisant les axiomes suivants.
	\begin{enumerate}
		\item
		$\bigpar{A, +, \times}$ est un anneau non nécessairement associatif : il est possible d'ajouter et de multiplier dans $A$. La loi $\times$ se distribue sur la loi $+$.
		\item
		$\bigpar{A, +}$ est un $\K$-espace vectoriel : il est possible de multiplier avec la loi $\cdot$ chaque élément de $A$ par un nombre de $\K$.
		Les lois sont associatives et se distribuent.
		\item Les loi $\times$ et $\cdot$ sont compatibles :
		pour $a, b \in A$ et $x, y\in \K$, $(x\cdot a) \times (y\cdot b) = (xy)( a \times b)$.
	\end{enumerate}
% true but boring?

}{dfn:algebre-corps}

\nt{
	Pour évaluer $p(x) \in \K[x]$ en $x\in A$, une algèbre sur $\K$, il conviendra alors que $\bigpar{A, +, \times}$ soit un anneau associatif (comme l'est un anneau par définition), commutatif, et unitaire.
	C'est le cas de tous les exemples de \ref{ex:algebres}.
	
	L'algèbre $\R^3$ vue comme $\R$-espace vectoriel et munie du produit vectoriel interne $\times = \land$ où
		\[ u \land v = \begin{pmatrix} u_1 \\ u_2 \\ u_3 \end{pmatrix} \land \begin{pmatrix} v_1 \\ v_2 \\ v_3 \end{pmatrix} = \begin{pmatrix} u_2v_3 - u_3v_2\\ u_3v_1-u_1v_3 \\ u_1v_2-u_2v_1 \end{pmatrix} \]
	n'est ni associative ($u\land(v\land w) \neq (u\land v)\land w$), ni commutative ($u\land v = -v\land u$), ni unitaire ($\not\exists e \tq u\land e = u \forall u\in\R^3$).
	Géométriquement, le produit vectoriel de deux vecteurs $u, v$ orthonormaux donne un vecteur $u\land v$ tel que $\bigpar{ u, v, u\land v}$ soit une base orthonormée de sens direct ($\det = +1$ ou respectant la règle du tire-bouchon) de $\R^3$.
}

\exe{}{
	Considérons l'ensemble $\R^*_+$ muni de la loi $\circ$ suivante.
		\[ x \circ y = x^y. \]
	Est-ce que $\bigpar{\R^*_+, \circ}$ est un groupe ?
	
	Trouver l'ensemble $S\subseteq\R^*_+$ le plus petit possible tel que $\bigpar{S, \circ}$ soit un groupe.
}{exe:R-pow}{
	\begin{enumerate}
		\item
		La loi est interne.
		\item
		Notons $e\in\R^*_+$ un élément neutre candidat.
		Alors $e$ doit vérifier $e\circ x = x \circ e = x$ pour tout $x\in\R^*_+$.
		
		L'égalité $x^e = x$ en $x=2$ implique $e=1$.
		L'égalité $e^x = x$ devient alors $1 = x$, ce qui n'est bien sûr pas vrai pour tout $x\in\R^*_+$ !
		
		La loi n'admet donc pas d'élément neutre.
		\item

			\[ x^{y^z} = (x^y)^z = x^{yz}, \]
		qui n'est pas vraie non plus pour tout $x, y, z\in\R^*_+$ (prendre par exemple $x=y=z=10$).
	\end{enumerate}

	L'ensemble $S = \{ 1 \}$ forme un groupe trivial lorsque muni de l'opération $\circ$.
