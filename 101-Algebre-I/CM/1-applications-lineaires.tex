\subsection{Projection orthogonale}

\dfn{projection orthogonale}{
	$\proj_u(v)$.
}{dfn:proj-orth}

\thm{}{
	Soient $u$ et $v$ deux vecteurs non nuls de $\R^n$.
	Alors $\proj_u(v)$ est le vecteur le plus proche de $v$ appartenant à $\scal{u}$.
}{thm:orth-proj-closest}

\thm{}{
	Soient $u$ et $v$ deux vecteurs non nuls de $\R^n$.
	Alors 	
		\[ \proj_u(v) = \frac{\scal{u, v}}{\scal{u, u}} u. \]
	%L'application $\proj_u$ est donc linéaire.
}{thm:proj-orth-formula}

\exe{}{
	Soient $u = \pvec12$ et $v = \pvec{-3}{-2}$ dans $\R^2$.

	Soient $u = \pvec12$ et $v = \pvec{-3}{-2}$ dans $\R^2$.
	\begin{enumerate}
		\item
		Calculer $\proj_u(v)$.
		\item

		\item
		Calculer la matrice $Q$ de l'application $\proj_u$ dans la base canonique.
		\item
		Calculer $P^2, Q^2$ et $P^n, Q^n$ pour tout $n\in\N$.
	\end{enumerate}

\exe{difficulty=1}{
	Soient $b_1^*, b_2, b_3 \in\R^3$ trois vecteurs linéairement indépendants.
	\begin{enumerate}
		\item
		Montrer que 
			\[ b_2^* = b_2 - \frac{\japb{b_2, b_1^*}}{\japb{b_1^*, b_1^*}} b_1^* \]
		est orthogonal à $b_1^*$.
		\item
		Montrer que 
			\[ b_3^* = b_3 - \frac{\japb{b_3, b_1^*}}{\japb{b_1^*, b_1^*}} b_1^* - \frac{\japb{b_3, b_2^*}}{\japb{b_2^*, b_2^*}} b_2^* \]
		est orthogonal à $b_1^*$ et $b_2^*$.
	\end{enumerate}
}{exe:gram-schmidt}{
	todo
}

\exe{difficulty=1}{

		\item
		Montrer l'identité de Pythagore :
			\[ \norm{u_1 + \cdots + u_n}^2 = \norm{u_1}^2 + \cdots + \norm{u_n}^2. \]
	\end{enumerate}
}{exe:ortho-is-free}{

\exe{difficulty=1}{
	Soit $u_1, \dots, u_n$ une famille orthonormée de $\R^n$ et $v\in\R^n$ un vecteur quelconque.
	Nous admettons que $v$ s'écrit comme combinaison linéaire 
		\[ v = \alpha_1 u_1 + \dots + \alpha_n u_n, \]
	avec $\alpha_i \in \R$ pour tout $i=1, \dots, n$.
	
	\begin{enumerate}
		\item
		Montrer que $\alpha_i = \japb{v, u_i}$ pour chaque $i=1, \dots, n$.
		\item
		Montrer que $\proj_{u_i}(v) = \alpha_i u_i$.
	\end{enumerate}
}{exe:ortho-coefficients}{
	todo
}
