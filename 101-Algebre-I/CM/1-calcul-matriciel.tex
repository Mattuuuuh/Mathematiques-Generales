\section{Matrices et vecteurs}

\dfn{matrice}{
	Soient $m, n\in\N^*$ deux entiers naturels.
	Une \emphindex{matrice $m\times n$ à coefficients réels} $M$ est un tableau rectangulaire de $m$ lignes et $n$ colonnes :
		\[ M = \begin{pmatrix} M_{11} & M_{12} & \dots & M_{1n} \\M_{21} & M_{22} & \dots & M_{2n} \\ \vdots & \vdots & \ddots & \vdots \\ M_{m1} & M_{m2} & \dots & M_{mn} \end{pmatrix}. \]
	Les coefficients $M_{ij}\in\R$ sont des nombres réels.
}{dfn:matrice}

\nomen{
	Les nombres $m\times n$ sont la \emphindex{taille} de la matrice.
	
	Si $m=n$, la matrice est dit \emphindex{matrice carrée}.
}

\notation{
	L'ensemble des matrices $m\times n$ à coefficients réels est noté $M_{m, n}(R)$ ou $\R^{m\times n}$.
}

\nt{
	Les indices et dimensions se lisent toujours « ligne puis colonne ».
	Une matrice $M \in \R^{4\times3}$ a donc $4$ lignes et $3$ colonnes.
	Le coefficient$M_{13}$ est situé ligne $1$, colonne $3$.
}

\nomen{
	Un vecteur $v\in\R^n$ est vu comme une matrice $n\times1$. Autrement dit, $\R^n = \R^{n\times1}$.
	Le vecteur est dit \emphindex{vecteur colonne} ou \emphindex{vecteur debout}.
	
	Une matrice $u \in \R^{1 \times n}$ est un \emphindex{vecteur ligne} ou \emphindex{vecteur couché}.
}

\subsection{Opérations sur les matrices}

\dfn{multiplication par un scalaire}{
	Soient $M\in\R^{m\times n}$ une matrice et $\alpha\in\R$ un réel appelé \emphindex{scalaire}.
	La multiplication $\alpha \cdot M$ distribue $\alpha$ sur chaque coefficient de $M$.
	\[ \alpha \cdot \begin{pmatrix} M_{11} & M_{12} & \dots & M_{1n} \\M_{21} & M_{22} & \dots & M_{2n} \\ \vdots & \vdots & \ddots & \vdots \\ M_{m1} & M_{m2} & \dots & M_{mn} \end{pmatrix} = \begin{pmatrix} \alpha M_{11} & \alpha M_{12} & \dots & \alpha M_{1n} \\\alpha M_{21} & \alpha M_{22} & \dots & \alpha M_{2n} \\ \vdots & \vdots & \ddots & \vdots \\ \alpha M_{m1} & \alpha M_{m2} & \dots & \alpha M_{mn} \end{pmatrix}. \]
}{dfn:mult-mat-scal}

\dfn{addition des matrices}{
	Soient $A, B\in\R^{m\times n}$ deux matrices \underline{de même taille}.
	L'\emphindex{addition matricielle} $A+B$ ajoute les coefficients de même position ensemble.
		\[ \begin{pmatrix} A_{11} & A_{12} & \dots & A_{1n} \\A_{21} & A_{22} & \dots & A_{2n} \\ \vdots & \vdots & \ddots & \vdots \\ A_{m1} & A_{m2} & \dots & A_{mn} \end{pmatrix} + \begin{pmatrix} B_{11} & B_{12} & \dots & B_{1n} \\B_{21} & B_{22} & \dots & B_{2n} \\ \vdots & \vdots & \ddots & \vdots \\ B_{m1} & B_{m2} & \dots & B_{mn} \end{pmatrix} = \begin{pmatrix} A_{11}+B_{11} & A_{12}+B_{12} & \dots & A_{1n}+B_{1n} \\ A_{21}+B_{21} & A_{22}+B_{22} & \dots & A_{2n}+B_{2n} \\ \vdots & \vdots & \ddots & \vdots \\ A_{m1}+B_{m1} & A_{m2}+B_{m2} & \dots & A_{mn}+B_{mn} \end{pmatrix}. \]
}{dfn:add-mat-mat}

\dfn{multiplication des matrices}{
	Soient $A\in\R^{m\times n}$ et $B\in\R^{n\times p}$ deux matrices.
	Le nombre de colonnes de $A$ est égal au nombre de lignes de $B$.
	
	Le \emphindex{produit matriciel} $A\cdot B$ est une matrice $m\times p$ telle que
		\[ (AB)_{ij} = \sum_{k=1}^{n} A_{ik} B_{jk}. \]
}{dfn:mult-mat-mat}

\nt{
	Le coefficient $(AB)_{ij}$ est égal au produit scalaire de la ligne $i$ de $A$ par la colonne $j$ de $B$.

	La multiplication matricielle respecte une sorte de « relation de Chasles » des tailles et des indices :
		\begin{align*}
			(m \times n) \cdot (n \times p) = (m \times p) && \et && (AB)_{ij} = \sum_{k=1}^{n} A_{ik}B_{kj}.
		\end{align*}
}

\motiv{
	La multiplication matricielle est définie de sorte à ce qu'une application linéaire de $\R^n$ dans $\R^m$ soit égale à
		\[ f(x) = A\cdot x, \]
	pour $A\in\R^{m\times n}$ une certaine matrice.
	Ceci généralise les applications linéaire de $\R$ dans $\R$ qui s'écrivent $f(x) = ax$ pour $a\in\R$ un certain nombre réel.
	
	De plus, si $g(x) = Bx$ est une application linéaire de $\R^p$ dans $\R^n$ (c'est-à-dire $B\in\R^{n\times p}$), alors
		\[ f(g(x)) = A(Bx) = (AB) x, \]
	par l'associativité démontrée section \ref{subsec:assoc}.
	
	La multiplication matricielle traduit donc naturellement les fonctions linéaires ainsi que leur composition.
}

\prop{non commutativité du produit matriciel}{
	Le produit matriciel n'est pas commutatif en général.
	Il existe deux matrices carrées $A, B\in\R^{n\times n}$ telles que, 
		\[ AB \neq BA \]
}{prop:prod-mat-mat-non-comm}

}{exe:mult-mat-mat-non-comm}{

\subsection{Matrice identité et matrices diagonales}

\notation{
	Lorsqu'un matrice est décrite, les coefficients $0$ peuvent être remplacés par des blancs.
	Par exemple,
		\[ \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix} = \begin{pmatrix} 1 & \\ & 1 \end{pmatrix}. \]
}

\dfn{matrice identité}{
	Soit $n\in\N^*$ un entier naturel.
	La \emphindex{matrice identité} de taille $n$ est donnée par
		\[ I_n = \begin{pmatrix} 1 & & & \\ & 1 & & \\ & & \ddots & \\ & & & 1 \end{pmatrix} \in \R^{n \times n}. \]
}{dfn:mat-id}

\notation{
	Lorsqu'un matrice est décrite, les coefficients $*$ peuvent être remplacés par coefficients réels quelconques.
	Par exemple,
		\[ \begin{pmatrix} * & 0 \\ 0 & * \end{pmatrix} = \begin{pmatrix} 2 & \\ & 6 \end{pmatrix} \ou \begin{pmatrix} -3 & \\ & 1 \end{pmatrix}. \]
}

\dfn{matrice diagonale}{
	Soit $n\in\N^*$ un entier naturel.
	Une \emphindex{matrice diagonale} de taille $n$ prend la forme
		\[ \begin{pmatrix} * & & & \\ & * & & \\ & & \ddots & \\ & & & * \end{pmatrix} \in \R^{n \times n}. \]
}{dfn:mat-diag}

\prop{action d'une matrice diagonale}{
	Soit $n\in\N*$, $D\in\R^{n\times n}$ une matrice diagonale, et $M\in\R^{n\times n}$ une matrice carrée quelconque.
		\[ D = \begin{pmatrix} d_1 & & & \\ & d_2 & & \\ & & \ddots & \\ & & & d_n \end{pmatrix} \in \R^{n \times n}. \]
	Alors 
	\begin{enumerate}
		\item
		Le produit $D\cdot M$ multiplie chaque ligne $i$ de $M$ par le coefficient $d_i$ correspondant.
		\item
		Le produit $M\cdot D$ multiplie chaque colonne $i$ de $M$ par le coefficient $d_i$ correspondant.
	\end{enumerate}
}{prop:mult-mat-diag}

\nt{
	Phrases récapitulatives à retenir.
	\begin{center}
		« Multiplier à gauche agit sur les lignes. »
		
		« Multiplier à droite agit sur les colonnes. »
	\end{center}
	L'exemple de non-commutativité de l'exercice \ref{exe:mult-mat-mat-non-comm} devient alors clair sans calcul.
}

	Par exemple,
		\[
		\begin{pmatrix} 1 & 2 & 3 \\ 4 & 5 & 6 \\ 7 & 8 & 9 \end{pmatrix} 
		=

\section{Propriétés des matrices}

\subsection{Associativité}
\label{subsec:assoc}

\prop{}{
	Le produit matriciel est associatif.
}{prop:matrix-assoc}

\pf{}{
	Soient $A\in\R^{m\times n}, B\in\R^{n\times p}, C\in\R^{p\times q}$ trois matrices à coefficients réels.
	Il s'agit de montrer que $A(BC) = (AB)C$, en tant que matrices $m\times q$.
	Pour cela, nous allons montrer que $[A(BC)]_{ij} = [(AB)C]_{ij}$ pour tout $i=1, \dots, m$ et $j=1, \dots q$.
	
	D'un part,
	\begin{align*}
		[A(BC)]_{ij} &= \sum_{k=1}^{n} A_{ik} (BC)_{kj}, \\
				&= \sum_{k=1}^{n} A_{ik} \sum_{l=1}^{p} B_{kl} C_{lj}, \\
				&= \sum_{k=1}^{n}\sum_{l=1}^{p} A_{ik} B_{kl} C_{lj}.\\
	\end{align*}
	D'autre part,
	\begin{align*}
		[(AB)C)]_{ij} &= \sum_{l=1}^{p} (AB)_{il} C_{lj}, \\
				&= \sum_{l=1}^{p} \sum_{k=1}^{n} A_{ik} B_{kl} C_{lj}, \\
				&= \sum_{k=1}^{n}\sum_{l=1}^{p} A_{ik} B_{kl} C_{lj},\\
	\end{align*}
	par permutation des sommes finies.
}

\cor{}{
	La puissance d'une matrice carrée $A$
		\[ A^n = \underbrace{A \cdot A \cdot \cdots \cdot A}_{\text{$n$ fois}} \]
	a un sens.
}{cor:Apown}

\notation{
	Par analogie aux nombres réels, nous posons $A^0 = I$ pour une matrice carrée $A$.
}

\subsection{Inverse}

\prop{}{
	Soit $M\in\R^{n \times n}$ une matrice où $n\in\N^*$.
	Supposons que $M$ admette un inverse à gauche : $M^{-G} M = I$.
	
	Alors l'inverse à gauche est unique et est également un inverse à droite : $M^{-G} = M^{-D} = M^{-1}$.
}{prop:inv}

\nomen{
	Une matrice admettant un inverse est dite \emphindex{matrice inversible}.
}

\pf{}{
	La preuve nécessite l'échelonnage des matrices du chapitre \ref{chap:forme-echelonnee}.
	La proposition sera alors réécrite (voir \ref{prop:inv2}) et démontrée.
}

\cor{}{
	Soit $A\in\R^{n\times n}$ une matrice carrée.
	Montrer l'équivalence suivante
	\begin{align*}
		\bigpar{\text{$A$ est inversible}} && \iff && \bigpar{\text{pour tout $b\in\R^n$, le système $Ax = b$ admet une unique solution $x\in\R^n$}}
	\end{align*}
}{cor:inv-sol}

\pf{}{
	\underline{Direction $\implies$.}
	
	Si $A$ est inversible, alors 
		\[ Ax = b \iff A^{-1} (Ax) = A^{-1} b \iff (A^{-1} A) x = A^{-1}b \iff I_n x = A^{-1} b \iff x = A^{-1}b. \]
	La solution existe et est unique.
	
	\underline{Direction $\impliedby$.}
	
	Nous souhaitons créer la matrice inverse $A^{-1}$ pour montrer que $A$ est inversible.
	Il s'agit donc de trouver une matrice $B\in\R^{n\times n}$ telle que $AB = I$.
	En notant $b_1, \dots, b_n \in\R^n$ les colonnes de $B$, l'exercice \ref{exe:mult-mat-mat2} implique que l'équation est équivalente aux systèmes
		\begin{align*}
			Ab_1 = e_1, && Ab_2 = e_2, && \dots, && Ab_n = e_n, 
		\end{align*}
	où $e_1, e_2, \dots, e_n$ sont les colonnes de $I_n$.
	
	L'existence de ces $b_i$ est garantie par hypothèse. La matrice $B$ vérifie donc $AB = I$ ce qui en fait l'inverse de $A$ d'après la proposition \ref{prop:inv}.
}

\nomen{
	L'ensemble des colonnes $e_1, e_2, \dots, e_n$ de la matrice identité $I_n$ est aussi appelé \emphindex{base canonique}\footnotemark.
}

\footnotetext{L'adjectif « canonique » signifie ici « naturelle ». C'est un choix privilégié pour sa simplicité. }

\subsection{Transposition}

\dfn{transposition}{
	La \emphindex{transposition} d'une matrice $M\in\R^{m\times n}$ est une matrice $M^T \in \R^{m\times n}$ vérifiant
		\[ M^T_{ij} = M_{ji}. \]
}{dfn:transposition}

\notation{
	La transposée est également notée $^tM$.
}

\nt{
	Chaque ligne de $M$ devient une colonne de $M^T$ (et inversement).
}

\ex{}{
	\[ \begin{pmatrix} 1 & 2 & 3 \\ 4 & 5 & 6 \end{pmatrix}^T = \begin{pmatrix} 1 & 4 \\ 2 & 5 \\ 3 & 6 \end{pmatrix}. \]
}{ex:transposition}


\nt{
	Le produit scalaire $\japb{u, v} = u_1 v_1 + \dots + v_n u_n$ de deux vecteurs colonne $u, v\in\R^n$ s'écrit donc de façon matricielle comme
		\[ \japb{u, v} = u^T v. \]
}

%\nomen{
%	Un vecteur $v \in \R^n$ est aussi appelé \emphindex{vecteur colonne}.
%	Sa transposée $v^T\in\R^{1\times n}$ est un \emphindex{vecteur ligne}.
%}

\prop{transposition du produit}{
	Soient $A \in \R^{m\times n}$ et $B\in\R^{n \times p}$. 
	Alors
		\[ (AB)^T = B^T A^T. \]
}{prop:transpos-produit}

\pf{}{
	Notons $a_i$ la ligne $i$ de $A$ et $b_j$ la colonne $j$ de $B$.
	Alors $(AB)_{ij} = \japb{a_i, b_j}$.
	
	Il suit donc que, d'une part, $(AB)^T_{ji} = \japb{a_i, b_j}$.
	
	D'autre part, $(B^T A^T)_{ji }$ est le produit scalaire de la ligne $j$ de $B^T$ avec la colonne $i$ de $A^T$.
	Par transposition et symétrie du produit scalaire, ce coefficient est égal au produit scalaire de la ligne $i$ de $A$ par la colonne $j$ de $B^T$.
	Donc $(B^T A^T)_{ji } = \japb{a_i, b_j} = (AB)^T_{ji}$, ce qui conclut.
}

\dfn{matrice symétrique}{
	Une matrice $M \in \R^{n\times n}$ est \emphindex{matrice symétrique} dès que $M^T = M$.
}{dfn:matrice-symetrique}
