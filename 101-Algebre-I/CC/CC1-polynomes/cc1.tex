\exe{3}{
	Montrer que la famille de trois polynômes $\bigset{ 3x^2 - 2x + 5 ;  5 ; 7x - 1}$ est linéairement indépendante sur $\R$.

	\fillwithlines{8cm}
}{exe:poly-lin-indep}{
	Il s'agit de montrer que la seule combinaison linéaire nulle est la combinaison triviale.
	Soient $\alpha, \beta, \gamma\in\R$ trois réels tels que
		\[ \alpha(3x^2 -2x+5) + \beta(5) + \gamma(7x-1) = 0. \]
	Alors $(3\alpha)x^2 + (7\gamma-2\alpha)x + (5\alpha+5\beta-\gamma) = 0$.
	D'après l'indépendance linéaire de la famille $\{ 1 ; x ; x^2 \}$, nous obtenons le système
		\[ \systeme[sort={\alpha,\beta, \gamma}]{3\alpha = 0, \\ 7\gamma-2\alpha = 0, \\ 5\alpha+5\beta-\gamma = 0.} \]
	La première équation donne $\alpha=0$, la deuxième $\gamma=0$ et la troisième $\beta=0$, ce qui conclut.
}



\exe{2}{
	Énoncer, \underline{sans le démontrer}, le théorème d'existence et d'unicité de la division euclidienne dans $\R[x]$.
	
	\fillwithlines{3cm}
}{exe:div-eucl}{
	Soient $f, g\in\R[x]$ avec $g\neq0$.
	Il existe d'uniques polynômes $q, r\in\R[x]$ tels que
		\[ f = gq + r, \]
	où $\deg(r) < \deg(g)$.
}

	\fillwithlines{5cm}

\exe{3}{
	En admettant le théorème de division euclidienne, démontrer le théorème \ref{thm:1} vu en classe.
	
	\thm{}{
		Soient $p\in\R[x]$ un polynôme à coefficient réels et $\alpha\in\R$ une racine réelle de $p$.
		Alors $x-\alpha$ divise $p$ dans $\R[x]$.
	}{thm:1}
	
	\fillwithlines{5cm}
}{exe:fa1}{
	La division euclidienne de $p$ par $x-\alpha$ donne
		\[ p(x) = (x-\alpha)q(x) + r(x), \]
	où $\deg(r) < \deg(x-\alpha) = 1$.
	Par conséquent, $\deg(r) \leq 0$ et $r$ est constant.
	En $x=\alpha$, $0 = 0 + r(\alpha)$, donc $r = 0$.
}

\exe{3}{
	Montrer que $x^2 - 1$ divise $x^{2026} - 1$ dans $\R[x]$ à l'aide du théorème \ref{thm:1}.
	
	\fillwithlines{5cm}
}{exe:x2-m-1}{
	$1$ est une racine de $x^{2026}-1$ donc
		\[ x^{2026}-1 = (x-1)q(x) \]
	pour un certain $q\in\R[x]$.
	$-1$ est racine de $x^{2026}-1$ donc
		\[ 0 = -2 q(-1), \]
	et $-1$ est aussi racine de $q$.
	Donc
		\[ x^{2026}-1 = (x-1)(x+1)\tilde{q}(x) = (x^2 - 1)\tilde{q}(x) \]
	pour un certain $\tilde{q}\in\R[x]$.
}


\exe{3}{
	Montrer que si $\alpha\in\R$ est une racine de $p\in\R[x]$ de multiplicité 2, alors $\alpha$ est une racine du polynôme dérivé $p'$.
	
	\fillwithlines{5cm}
}{exe:root-of-deriv}{
	Le polynôme $(x-\alpha)^2$ divise $p$, donc $p(x) = (x-\alpha)^2 q(x)$ pour un certain $q\in\R[x]$.
	
	En dérivant, nous obtenons
		\[ p'(x) = \left[  (x-\alpha)^2 q(x) \right]' = 2(x-\alpha)q(x) + (x-\alpha)^2q'(x) = (x-\alpha)\left[ 2q(x) + (x-\alpha)q'(x) \right]. \]
}
