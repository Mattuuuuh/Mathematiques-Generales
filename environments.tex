		{\Large \bfseries TABLE DES MATIÈRES \bigskip}

	\begin{multicols}{2}\setlength{\columnseprule}{1pt}
		\titleformat{\chapter}{}{}{0pt}{} 
		\titlespacing*{\chapter}{0pt}

		\tableofcontents
	\end{multicols}
	\hrule
}
